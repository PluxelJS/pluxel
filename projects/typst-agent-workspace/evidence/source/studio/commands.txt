from pathlib import Path
from pypdf import PdfReader
import hashlib, json
root = Path('projects/typst-agent-workspace/evidence/source/studio')
evidence = json.loads((root / 'pdf-inspection.json').read_text())
for name, expected in evidence['files'].items():
    path = root / name
    reader = PdfReader(path)
    text = '\n'.join(page.extract_text() for page in reader.pages)
    assert len(reader.pages) == expected['pages'] == 1
    assert all(value in text for value in expected['requiredText'])
    assert hashlib.sha256(path.read_bytes()).hexdigest() == expected['sha256']
    assert [hashlib.sha256(page.get_contents().get_data()).hexdigest()
            for page in reader.pages] == expected['pageContentSha256']
assert evidence['reopenedPdfTextMatches']
assert evidence['reopenedPdfContentStreamsMatch']
print('Five PDFs: text, page count, file and decoded content hashes verified.')
